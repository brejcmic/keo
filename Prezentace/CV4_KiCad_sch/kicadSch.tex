%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Globální a projektové knihovny}
	\begin{description}
		\item[Globální knihovny]: \\ Jsou dostupné ve všech projektech uživatele na daném počítači.
		\item[Projektové knihovny]: \\ Jsou dostupné pouze v právě otevřeném projektu.
	\end{description}

	\vfill
	\begin{block}{Doporučení}
		Vlastní knihovnu určenou pro konkrétní návrh uložte do složky projektu a přidejte ji jako projektovou. Projekt pak lze snáze přenést na jiný počítač.
	\end{block}

\end{frame}

%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Přidání knihovny symbolů}
	\begin{enumerate}
		\item Otevřete kartu \textbf{Globální knihovny} nebo \textbf{Projektové knihovny}.
		\item Klikněte na tlačítko \textbf{Přidat existující knihovnu}.
		\item Vyberte soubor knihovny, obvykle \texttt{*.kicad\_sym}.
		\item Zkontrolujte přezdívku a cestu knihovny.
		\item Potvrďte tlačítkem \textbf{OK}.
	\end{enumerate}

	\vfill
	\begin{block}{Pozor}
		Zápis v tabulce knihoven odkazuje na soubor knihovny. Samotný soubor se tím do projektu nekopíruje.
	\end{block}

\end{frame}

%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Tabulky knihoven}
	\begin{description}
		\item[sym-lib-table] Seznam přezdívek a cest ke knihovnám symbolů.
		\item[fp-lib-table] Seznam přezdívek a cest ke knihovnám pouzder.
	\end{description}

	\medskip
	Každý z těchto souborů může existovat na dvou místech:
	\begin{itemize}
		\item v konfigurační složce KiCadu pro \textbf{globální} knihovny,
		\item ve složce projektu pro \textbf{projektové} knihovny.
	\end{itemize}

	\vfill
	Soubory \texttt{*-lib-table} nejsou knihovny; pouze říkají KiCadu, kde knihovny hledat.

\end{frame}

%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Správa knihoven pouzder}
	Postup je stejný jako u knihoven symbolů:
	\begin{enumerate}
		\item V editoru DPS zvolte \textbf{Nastavení → Spravovat knihovny pouzder\ldots}
		\item Vyberte globální nebo projektovou kartu.
		\item Přidejte existující knihovnu pouzder, obvykle složku \texttt{*.pretty}.
		\item Potvrďte tlačítkem \textbf{OK}.
	\end{enumerate}

	\vfill
	Jednotlivá pouzdra KiCadu jsou soubory \texttt{*.kicad\_mod}; jejich umístění eviduje tabulka \texttt{fp-lib-table}.

\end{frame}

\begin{frame}
	\frametitle{Okno nastavení schématu}
	V editoru schématu zvolte \textbf{Soubor → Nastavení schématu\ldots}
	nebo použijte tlačítko nastavení v horní liště.

	\medskip
    \centering
			\includegraphics[width=\linewidth,height=0.68\textheight,keepaspectratio]{obr/schematic_editor_settings.png}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Obecná nastavení}
	\begin{description}
		\item[Formátování:] Vzhled schématu: velikosti textů, šířky čar, spojovací body, přemostění a odkazy mezi listy.
		\item[Anotace:] Způsob automatického přidělování referencí součástek, např. \texttt{R1}, \texttt{C1} nebo \texttt{U1}.
		\item[Šablony názvů polí:] Vlastní pole automaticky přidávaná ke všem symbolům, např. výrobce nebo objednací číslo.
		\item[Přednastavení BOM:] Uložená nastavení obsahu a formátu kusovníku; vlastní předvolby se zde zobrazují a odstraňují.
	\end{description}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Pravidla a projektová nastavení}
	\begin{description}
		\item[Závažnost porušení:] Určuje, které kontroly ERC jsou chyby, varování nebo se ignorují.
		\item[Mapa konfliktů vývodů:] Nastavuje povolená a chybná spojení elektrických typů vývodů.
		\item[Třídy spojů:] Sdružují sítě (vodiče) se společnými grafickými vlastnostmi a pravidly návrhu DPS.
		\item[Definice aliasů sběrnice:] Pojmenované skupiny signálů používaných ve sběrnicích.
		\item[Textové proměnné:] Projektové hodnoty opakovaně použitelné v textech a polích.
		\item[Vložené soubory:] Soubory uložené přímo ve schématu pro snadnější přenos projektu.
	\end{description}
\end{frame}

				\item šířku vodiče a sběrnice,
				\item šířku cesty, izolační vzdálenost a rozměry prokovů na DPS.
			\end{itemize}

%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Vytvoření a přiřazení třídy spojů}
	\begin{enumerate}
		\item V části \textbf{Třídy spojů} přidejte novou třídu a pojmenujte ji, např. \texttt{POWER}.
		\item Nastavte její vlastnosti a případně pořadí priority.
		\item Síť přiřaďte jedním ze způsobů:
		\begin{itemize}
			\item vzorem názvu sítě, např. \texttt{+5V} nebo \texttt{USB*},
			\item pravým tlačítkem na pojmenované síti → \textbf{Přiřadit třídu spojů\ldots},
			\item direktivním označením nebo polem \texttt{Net Class} u označení sítě.
		\end{itemize}
	\end{enumerate}

	Třída \texttt{Default} je vždy přítomná a doplňuje vlastnosti, které jiná třída neurčuje.

%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Použití textových proměnných}
	Příklad projektové proměnné:
	\begin{center}
		\texttt{VERSION = 1.0}
	\end{center}
	Text \texttt{Verze: \$\{VERSION\}} se zobrazí jako \texttt{Verze: 1.0}.

	\medskip
	Proměnné lze použít například v:
	\begin{itemize}
		\item grafických textech schématu i DPS,
		\item polích symbolů, pouzder, označení a hierarchických listů,
		\item popisovém poli výkresu.
	\end{itemize}

	\vfill
	Projektové proměnné jsou společné pro editor schématu i editor DPS. Změna hodnoty se automaticky projeví na všech místech použití.
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Vestavěné textové proměnné}
	KiCad nabízí také systémové proměnné, například:
	\begin{description}
		\item[\$\{TITLE\}:] název výkresu,
		\item[\$\{COMPANY\}:] název organizace,
		\item[\$\{COMMENT1\}:] první komentář výkresu,
		\item[\$\{FILENAME\}:] název souboru kořenového schématu,
		\item[\$\{CURRENT\_DATE\}:] aktuální datum,
		\item[\$\{\#\}:] číslo aktuálního listu,
		\item[\$\{\#\#\}:] celkový počet listů schématu.
	\end{description}

	\vfill
	Dostupnost systémových proměnných závisí na místě, kde jsou použity.
\end{frame}

\begin{frame}
	\frametitle{Položky nastavení stránky}
  \small
	\begin{description}

		\item[Soubor výkresového listu:] Volitelná vlastní šablona rámečku a razítka.

%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Číslování revize zařízení}
	Pro cvičení použijeme zápis \textbf{X.Y.Z}, například \textbf{2.1.3}:

	\medskip
	\begin{description}
		\item[První číslice -- X:] Konektorová, funkční, signálová a mechanická kompatibilita mezi DPS.
		\item[Druhá číslice -- Y:] Označuje významnější úpravu zařízení při zachování kompatibility.
		\item[Třetí číslice -- Z:] Volitelně označuje menší úpravu bez změny funkce a rozhraní (oprava potisku, změna filtračního kondenzátoru apod.).
	\end{description}

	\vfill
	\begin{block}{Pravidlo kompatibility}
		DPS se stejným prvním číslem revize musí zachovat možnost vzájemné náhrady.
	\end{block}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Kdy změnit číslo revize}
	\begin{description}
		\item[První číslice:] Změna konektorů, jejich zapojení, napájení nebo chování, která znemožní přímou náhradu starší DPS. Patří sem i změna rozměrů a montážních otvorů.
		\item[Druhá číslice:] Nová funkce, změna integrovaného obvodu, doplnění vizuální signalizace nebo jiná úprava zařízení při zachování možnosti přímé náhrady.
		\item[Třetí číslice:] Optimalizace propojení, drobná oprava potisku, posun součástky nebo jiná malá výrobní úprava.
	\end{description}

	\vfill
	\begin{exampleblock}{Příklad}
		Revize \textbf{2.0}, \textbf{2.1} a \textbf{2.1.1} jsou vzájemně nahraditelné. Revize \textbf{3.0} již může používat nekompatibilní rozhraní.
	\end{exampleblock}

	\smallskip
	KiCad význam čísel neurčuje; pole \textbf{Revize} je volný text a pravidla revizí stanovuje projekt.
\end{frame}

\begin{frame}

	Najeďte kurzorem na symbol nebo jej označte:
	\begin{description}
		\item[V:] upraví pouze pole \textbf{Hodnota},
		\item[F:] upraví pouze pole \textbf{Pouzdro},

		Pouzdro musí odpovídat skutečné součástce: počtu vývodů, číslování vývodů, jejich rozteči i fyzickým rozměrům.
	\end{block}

%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Více prvků v jednom pouzdře}
	Některé integrované obvody obsahují několik stejných funkčních jednotek.

	\medskip
	Příklad \textbf{74HC14}:
	\begin{itemize}
		\item jednotky \textbf{A--F}: šest samostatných invertorů,
		\item jednotka \textbf{G}: společné napájecí vývody,
		\item všechny jednotky patří k jednomu pouzdru a mají společnou referenci, např. \textbf{U1A--U1G},
		\item hodnota a pouzdro jsou společné pro celý obvod \textbf{U1}.
	\end{itemize}

	\vfill
	\begin{alertblock}{Důležité}
		Do schématu vložte také napájecí jednotku. Jinak nebudou napájecí vývody integrovaného obvodu zapojeny.
	\end{alertblock}
\end{frame}

\begin{frame}
	\frametitle{Vložení víceprvkové součástky}
	\begin{enumerate}
		\item Stiskněte \textbf{A} a vyberte \texttt{74HC14}.
		\item Zapněte volbu \textbf{Umístit všechny jednotky}.
		\item Umístěte postupně jednotky \textbf{A--F} a napájecí jednotku \textbf{G}.
		\item Nepoužité invertory ponechte ve schématu a jejich vstupy zapojte do definované logické úrovně.
	\end{enumerate}

	\medskip
	Pokud nechcete vložit všechny jednotky najednou:
	\begin{itemize}
		\item umístěte první jednotku,
		\item v kontextové nabídce symbolu použijte \textbf{Umístit další jednotku symbolu},
		\item KiCad nabídne další dosud neumístěnou jednotku se stejnou referencí.
	\end{itemize}

	\vfill
	Klávesa \textbf{Esc} kdykoli ukončí posloupnost vkládání.

\end{frame}

\begin{frame}
	\frametitle{Skupinová sběrnice}

	\begin{center}

	\end{center}

\begin{frame}
	\frametitle{Typy vývodů hierarchického listu}
	Směr se uvádí z pohledu podřízeného listu:
	\begin{description}
		\item[Vstup:] Signál vstupuje z rodičovského listu do podřízeného bloku.

		\item[Obousměrný:] Signál může proudit oběma směry, např. datová linka.
		\item[Třístavový:] Výstup může kromě logických úrovní přejít do vysoké impedance.
		\item[Pasivní:] Směr není určen; vhodné pro pasivní nebo obecné propojení.
	\end{description}

	\begin{enumerate}
		\item V podřízeném listu vytvořte hierarchický popisek \texttt{DATA[0..7]}.

		\item K vývodu v rodičovském listu připojte sběrnici.
		\item Uvnitř listu sběrnici rozviňte na jednotlivé signály \texttt{DATA0} až \texttt{DATA7}.

%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Co ERC kontroluje}
	\begin{itemize}
		\item nezapojené vývody, vodiče a popisky,
		\item vstupy bez zdroje signálu,
		\item napájecí vstupy bez napájecího výstupu,
		\item nepovolené kombinace vývodů, např. dva výstupy spojené dohromady,
		\item chyby hierarchických popisků, vývodů listů a sběrnic,
		\item duplicitní označení součástek a neanotované symboly,
		\item chybné nebo neexistující přiřazení pouzdra,
		\item překlepy v názvech sítí a další nekonzistence schématu.
	\end{itemize}

	\vfill
	\begin{alertblock}{Omezení kontroly}
		ERC ověřuje elektrická pravidla a konzistenci schématu, ale nezaručuje správnou funkci zapojení. Výsledek závisí na správně nastavených elektrických typech vývodů symbolů.
	\end{alertblock}
\end{frame}

				\item \textbf{Varování} -- vyžaduje kontrolu,

				\item Přidejte symbol klávesou \textbf{A}.
				\item Vyhledejte \texttt{PWR\_FLAG} a připojte jej k napájené síti.
				\item Znovu spusťte ERC.
			\end{enumerate}

	\end{columns}

	\medskip
	\begin{alertblock}{Důležité}
		\texttt{PWR\_FLAG} nevytváří napětí ani spojení. Pouze označí již napájenou síť pro ERC; umisťujte jej jen tam, kde je zdroj skutečně připojen.
	\end{alertblock}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Rozsah a pořadí číslování}
	\begin{description}
		\item[Rozsah:] Celé schéma, pouze aktuální list nebo pouze označené symboly.
		\item[Vnořené listy:] Volba \textbf{Zahrnout podřízené listy} rozšíří vybraný rozsah také do jeho hierarchie.
		\item[Pořadí X:] Symboly se číslují zleva doprava; levější symboly dostanou nižší čísla.
		\item[Pořadí Y:] Symboly se číslují shora dolů; horní symboly dostanou nižší čísla.
	\end{description}

	\vfill
	\begin{block}{Existující reference}
		\textbf{Zachovat existující anotaci} doplní pouze symboly s otazníkem. \textbf{Obnovit existující anotaci} přečísluje všechny symboly ve zvoleném rozsahu.
	\end{block}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}
	\frametitle{Volba čísel referencí}
	V části \textbf{Číslování} lze zvolit počáteční hodnotu:
	\begin{description}
		\item[První volné číslo:] použije nejnižší dostupné číslo větší než zadaná hodnota; obvykle se zadává \texttt{0}.
		\item[Číslo listu $\times$ 100:] na listu 3 vzniknou např. reference \texttt{R301}, \texttt{R302}, \ldots
		\item[Číslo listu $\times$ 1000:] na listu 3 vzniknou např. reference \texttt{R3001}, \texttt{R3002}, \ldots
	\end{description}

	\medskip
	\begin{itemize}
		\item \textbf{Vymazat anotaci} odstraní reference ve zvoleném rozsahu.
		\item \textbf{Očíslovat} provede zvolené přečíslování.
		\item Spodní část okna zobrazí provedené akce, varování a chyby; výpis lze uložit.
	\end{itemize}

	\vfill
	\begin{alertblock}{Pozor}
		Po přečíslování již existujícího návrhu aktualizujte DPS ze schématu, aby reference součástek a pouzder zůstaly shodné.
	\end{alertblock}
\end{frame}

		\item[Prostřední panel:] symboly schématu a jejich aktuálně přiřazená pouzdra.
		\item[Pravý panel:] pouzdra vyhovující zapnutým filtrům.
