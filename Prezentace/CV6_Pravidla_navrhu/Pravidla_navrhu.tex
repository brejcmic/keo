\begin{frame}[t]{Stackup: nejde pouze o počet vrstev}
	\begin{itemize}
		\item \textbf{Stackup} určuje pořadí měděných vrstev, jejich funkci a tloušťky izolačních materiálů mezi nimi.
		\item \textbf{SIG}: signály; \textbf{PWR}: napájení; \textbf{GND}: referenční vodič a návratová cesta proudu.
		\item Zápis \textbf{SIG + PWR} označuje jednu společnou vrstvu pro signálové trasy a rozvod napájení, nikoli dvě vrstvy.
		\item Pořadí v obrázcích čteme \textbf{shora dolů}. Čísla L1--L4 odpovídají pořadí měděných vrstev.
	\end{itemize}
	\begin{block}{Základní doporučení}
	Signál potřebuje blízkou a souvislou referenční plochu. Důležité jsou i rychlost hran signálu a délka vedení, nejen hodinová frekvence.
	\end{block}
	\vfill
	{\tiny Zdroj doporučení: [1, 5, 6]. Úplný seznam zdrojů je na konci prezentace.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Dvě vrstvy: (SIG + PWR) / GND}

	{\centering
		\includegraphics[width=0.7\linewidth]{obr/DPS_SG.eps}\par}
    
	\vspace{1cm}
	Signály a napájení veďte převážně na L1; protější L2 ponechte co nejvíce souvislou jako návratovou cestu GND.
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Dvě vrstvy: výhody a omezení}
  \textbf{Výhody:}
  \begin{itemize}
    \item jednoduchá a levná výroba,
    \item přehledná návratová cesta oproti desce s rozdělenou zemí na obou stranách.
  \end{itemize}

  \textbf{Nevýhody:}
	\begin{itemize}
		\item méně prostoru pro signály a napájení,
    \item u běžné tloušťky 1,5 mm je protější GND dost vzdálená vůči signálu $\Rightarrow$ horší vazba a případně řízení impedance.
	\end{itemize}
  
  \begin{block}{Doporučení}
	Použijte pro jednodušší obvody s krátkými vedeními. Spodní GND přerušujte co nejméně -- trasy přes celou spodní stranu mohou rozříznout návratovou cestu.
  \end{block}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Čtyři vrstvy se dvěma plochami GND}

	{\centering
		\includegraphics[width=0.7\linewidth]{obr/DPS_SGGS.eps}\par}
    
	\vspace{1cm}
	Napájení lze rozvádět na jedné nebo obou vnějších vrstvách. Dvě vnitřní GND plochy (L2 a L3) ponechte souvislé a vzájemně je propojte.

\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Dvě plochy GND: výhody a omezení}

  \textbf{Výhody:}
  \begin{itemize}
    \item obě signálové vrstvy mají blízkou referenční GND,
    \item vnitřní plochy oddělují signály na protilehlých stranách.
  \end{itemize}
  
  \textbf{Nevýhody:}
	\begin{itemize}
		\item napájení sdílí prostor se signály, není k dispozici samostatná PWR vrstva. Pro větší proudy a mnoho napěťových úrovní (zdrojů) může chybět místo.
	\end{itemize}
  
	\begin{block}{Doporučení}
	Vhodný výchozí stackup pro běžnou čtyřvrstvou digitální desku. GND plochy propojte a u přechodu signálu mezi L1 a L4 je vhodné vložit blízký GND prokov.
  \end{block}
\end{frame}

\begin{frame}[t]{Čtyři vrstvy: SIG / GND / PWR / SIG}

	{\centering
		\includegraphics[width=0.7\linewidth]{obr/DPS_SGPS.eps}\par}
    
	\vspace{1cm}
	L1 má referenci GND, zatímco nejbližší referencí pro L4 je PWR. Právě tento rozdíl určuje vrstvu L1 pro rychlé signály.

\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{SIG / GND / PWR / SIG: výhody a omezení}

  \textbf{Výhody:}
  
  \begin{itemize}
    \item dvě signálové vrstvy a samostatný rozvod napájení -- snazší vedení spojů,
    \item horní signály mají blízkou GND.
  \end{itemize}
  
  \textbf{Nevýhody:}
	\begin{itemize}
		\item spodní signály mají nejbližší referenci PWR. Rozdělení napájecí plochy a změna reference při přechodu vrstev komplikují návrat proudu.
	\end{itemize}
	
  \begin{block}{Doporučení}
	Rychlé signály veďte přednostně na L1 nad GND. Plná PWR může být referencí pro L4, ale je nutné vyřešit návaznost návratového proudu na GND (blokovací kondenzátory).
  \end{block}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Návratová cesta při změně signálové vrstvy}
	\begin{itemize}
		\item U rychlých změn proudu se návratový proud soustřeďuje do blízké referenční plochy pod trasou.
		\item \textbf{GND → GND:} při přechodu L1 $\leftrightarrow$ L4 propojte obě referenční plochy GND prokovem poblíž signálového prokovu.
		\item \textbf{GND → PWR:} samotný GND prokov nestačí. Přechod návratového proudu vyžaduje vhodnou vazbu mezi PWR a GND, např. blízký blokovací kondenzátor s krátkým připojením.
		\item Signál neveďte přes mezeru, výřez nebo rozhraní různých napětí v jeho referenční ploše.
	\end{itemize}
	\begin{block}{Výhoda dvou GND vrstev}
	Návratovou cestu lze při přechodu jednodušeji navázat GND prokovem. GND a PWR se naopak nesmějí spojit přímým prokovem.
	\end{block}
	\vfill
	{\tiny Zdroj doporučení: [4, 5, 7, 9] Peterson, Bogatin, Kay, Salmony.}
\end{frame}

\begin{frame}[t]{Tloušťky dielektrika a vrstvy napájení}
	\begin{itemize}
		\item U čtyřvrstvé desky požadujte blízké dvojice \textbf{L1--L2} a \textbf{L3--L4}; běžná stavba má tenké prepregy a silnější středové jádro.
		\item Menší vzdálenost signálu od reference pomáhá omezit plochu proudové smyčky. Impedanci ale určuje i šířka a tloušťka mědi a vlastnosti dielektrika.
		\item Vzdálené plochy PWR a GND mají malou vzájemnou kapacitu. \textbf{Samostatná PWR vrstva nenahrazuje blokovací kondenzátory.}
		\item Konkrétní tloušťky, materiály a dostupné varianty ověřte u výrobce \textbf{před návrhem impedančně řízených spojů}.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [2, 6] Z. Peterson, Phil's Lab.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Jak zvolit výchozí skladbu?}
	\small
	\begin{description}
		\item[(SIG + PWR) / GND] Pro jednoduché, méně husté desky. Prioritou je zachování GND, ne zaplnění obou stran signály.
		\item[SIG / GND / PWR / SIG] Když pomůže samostatná napájecí plocha a nejdůležitější signály zůstanou nad GND. Není to automaticky chybné řešení, ale spodní signály vyžadují pozornost.
		\item[(SIG + PWR) / GND / GND / (SIG + PWR)] Vhodný výchozí kompromis pro digitální signály na obou stranách. Napájení řešte dostatečně širokými trasami či polygony.
	\end{description}
	\begin{block}{Neexistuje jedna skladba pro všechny desky}
	Rozhodují proudy, napětí, rychlost hran, hustota zapojení a požadavky na rušení. Pokud se napájení a signály nevejdou bez porušení návratových cest, zvažte více vrstev.
	\end{block}
	\vfill
	{\tiny Zdroj doporučení: [1--6]. Výukové shrnutí; úplný seznam zdrojů následuje na konci.}
\end{frame}

\section{\texorpdfstring{Přerušení zpětné cesty}{Preruseni zpetne cesty}}
%------------------------------------------------------------------------------
\begin{frame}[t]{Funkční slot: deska není proříznutá, GND ano}
	\small
	\begin{itemize}
		\item Zde výrazem \textbf{funkční slot} označujeme mezeru v referenční mědi vytvořenou jinými prvky návrhu, nikoli mechanické frézování DPS.
		\item Vzniká například spojením \textbf{izolačních odstupů kolem cizích plošek} konektoru nebo vedením \textbf{jiné sítě uvnitř GND plochy}.
		\item Signál může být správně propojený, ale jeho vysokofrekvenční návrat musí mezeru obejít. Zvětšuje se proudová smyčka a riziko rušení i změny impedance.
	\end{itemize}
	\begin{block}{Sledujeme dva vodiče, ne jen signál}
	Průchod mezi piny ani křížení na různých vrstvách nejsou samy o sobě chybou. Rozhodující je, zda pod signálem zůstává dostatečně souvislá referenční měď.
	\end{block}
	\vfill
	{\tiny Zdroj doporučení: [1, 9, 10] Peterson, Salmony, Texas Instruments.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Konektor: signál projde, návrat musí oklikou}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.50\textheight,keepaspectratio]{obr/slot_connector_bad.pdf}\par}
	\begin{itemize}
		\item Trasa na F.Cu se vejde mezi piny. Odstupy cizích průchozích plošek však na B.Cu splývají do \textbf{souvislé mezery v GND}.
		\item Návrat nemůže sledovat signál přímo; musí obejít řadu plošek. Případně ponechané úzké hrdlo mědi může být také nevhodné z pohledu impedance.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [9, 10] P. Salmony, Texas Instruments.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Konektor: veďte signál nad souvislou GND}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.52\textheight,keepaspectratio]{obr/slot_connector_better.pdf}\par}
	\begin{itemize}
		\item Zde signál obchází konec řady plošek \textbf{nad souvislou GND}; návrat může zůstat blízko jeho trasy.
		\item Jinou možností je změnit rozmístění či pinout a přidělit vhodné piny GND. Výsledek posuzujte v \textbf{naplněné referenční ploše}, ne jen podle spojnic.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [9, 10] P. Salmony, Texas Instruments.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Dvě vrstvy: spodní trasa rozřízne GND}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.52\textheight,keepaspectratio]{obr/slot_crossing_bad.pdf}\par}
	\begin{itemize}
		\item Horní signál kříží v půdorysu jinou trasu na B.Cu. \textbf{Spodní trasa i její odstupy vytlačí GND} právě pod horním signálem.
		\item Dlouhá spodní trasa vynutí dlouhou okliku návratu. Pokud oddělí GND až od okraje k okraji, nemusí zbýt ani vodivá cesta v této vrstvě.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [1, 10] Z. Peterson (dvě vrstvy), TI (slot a proudová smyčka).}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Křížení: blízký GND můstek jako kompromis}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.50\textheight,keepaspectratio]{obr/slot_crossing_better.pdf}\par}
	\begin{itemize}
		\item Nejprve zkuste spodní trasu \textbf{přemístit nebo zkrátit} tak, aby nepřerušovala referenci důležitého signálu.
		\item Kompromis: blízké GND prokovy a krátký GND můstek na F.Cu přenesou návrat přes mezeru. \textbf{Nespojí cizí síť s GND}; stále přidávají indukčnost a nenahrazují souvislou plochu.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [1, 9, 10] Peterson, Salmony, Texas Instruments.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Křížení: kapacitní můstek jako další kompromis}
	\footnotesize
	{\centering
	\includegraphics[width=\linewidth,height=.39\textheight,keepaspectratio]{obr/slot_crossing_cap.pdf}\par}
	\begin{itemize}
		\item Kondenzátor na F.Cu spojí přes dva blízké prokovy \textbf{GND po obou stranách mezery}. Je těsně vedle křížení.
		\item \textbf{Vysokofrekvenční návrat přejde kapacitní vazbou}; DC přes kondenzátor neteče. Stejnosměrné propojení GND musí zůstat zajištěno jinudy.
		\item Volte nízkou impedanci ve frekvenčním pásmu signálu včetně jeho hran: rozhodují kapacita i ESL $\Rightarrow$ krátké přípoje. Pro tutéž GND bývá vhodnější přímý vodivý můstek.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [7, 10] TI: kapacitní vazba návratu, omezení slotů a impedance reálného C.}
\end{frame}

\section{\texorpdfstring{Zónování DPS}{Zonovani DPS}}
%------------------------------------------------------------------------------
\begin{frame}[t]{Zónování DPS: nejprve plán, potom trasy}
	\small
	\begin{itemize}
		\item \textbf{PCB partitioning / floor planning} je rozdělení obvodu na funkční bloky a plán jejich umístění na desce před podrobným trasováním. Zde používáme název \textbf{zónování DPS}.
		\item Sdružujeme součástky, které spolu elektricky úzce souvisejí: například vstupní filtr, zesilovač a jeho zpětnou vazbu. Zároveň oddělujeme zdroje rušení a tepla od citlivých obvodů.
		\item Cílem jsou \textbf{krátké místní vazby, malé proudové smyčky a přehledný tok signálů i výkonu}. Nestačí, že se součástky na desku vejdou.
	\end{itemize}
	\begin{block}{Co znamená zóna v části}
	Oblast pro určitou skupinu součástek.
	\end{block}
	\vfill
	{\tiny Zdroje: [10, 11, 22] TI, Analog Devices; plánování funkčních bloků.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Podle čeho součástky sdružovat a oddělovat?}
	\footnotesize
	\begin{tabular}{>{\raggedright\arraybackslash}p{.20\linewidth}|>{\raggedright\arraybackslash}p{.72\linewidth}}
		\textbf{Hledisko} & \textbf{Důsledek pro rozmístění}\\\hline
		Funkce a vazby & IC spolu s blokováním, zpětnou vazbou a souvisejícími pasivními prvky; příslušný konektor poblíž.\\[1ex]
		Výkon a proud & Výkonový stupeň u přívodu a zátěže, krátké silové cesty; slabé měřicí signály mimo jejich proudové smyčky.\\[1ex]
		Frekvence a hrany & Hodiny, rychlá rozhraní a spínací uzly držet místně; kritické spoje co nejkratší podle jejich požadavků.\\[1ex]
		Citlivost & Malé signály, reference a vysokoimpedanční vstupy chránit před kapacitní, induktivní i společnou impedanční vazbou.\\[1ex]
		Teplota & Zdroje ztrát potřebují cestu pro teplo; přesné obvody a čidla okolní teploty umístit mimo jejich tepelný vliv.
	\end{tabular}
  
	\vspace{5mm}
	\textbf{Hlediska se překrývají.} Měnič spíná, vede velký proud a hřeje; ADC propojuje citlivou analogovou část s digitálním rozhraním.
	\vfill
	{\tiny Zdroje: [10, 20, 22--24] TI, Analog Devices; vlastní souhrn návrhových hledisek.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Postup: od blokového schématu k rozmístění}
	\small
	\begin{enumerate}
		\item Vyznačte obrys, otvory, konektory, ovládání a omezení dané krabičkou. Vyberte prostor pro chladiče.
		\item Rozdělte schéma na funkční bloky dle specifických hledisek. Označte kritické vazby a zdroje rušení.
		\item Oblasti bloků navrhujte také podle přítomnosti konektorů a toku signálu. Vyhraďte místo pro výkonové cesty, odvod tepla a potřebné odstupy.
		\item Uvnitř bloků umístěte nejdříve kritické součástky: místní blokování, komutační smyčku měniče, vstupní filtr, referenci či krystal.
		\item Zkontrolujte, zda bloky půjde propojit \textbf{nad vhodnou referencí bez zbytečných oklik}. Pokud ne, upravte rozmístění ještě před trasováním.
	\end{enumerate}
	\vfill
	{\tiny Zdroje: [10, 16, 20, 22] vlastní postup vycházející ze zásad TI a Analog Devices.}
\end{frame}

\begin{frame}[t]{Příklad: promíchané bloky komplikují návrh}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.49\textheight,keepaspectratio]{obr/floorplan_bad.pdf}\par}
	\begin{itemize}
		\item Vstupní analogová cesta obchází desku, měnič je mezi měřicími bloky a silová cesta vede přes další oblasti.
		\item Dlouhé vazby a překrývající se koridory zvyšují riziko rušení a komplikují návraty. \textbf{Problém vznikl už při rozmístění.}
	\end{itemize}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Příklad: související bloky tvoří krátký řetězec}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.49\textheight,keepaspectratio]{obr/floorplan_good.pdf}\par}
	\begin{itemize}
		\item Čidlo $\rightarrow$ filtr a zesilovač $\rightarrow$ ADC $\rightarrow$ MCU. ADC je u rozhraní oblastí: analogové piny k analogu, digitální k MCU.
		\item Napájecí a výkonová část mají vlastní prostor. Větve napájení a jejich návraty navrhněte tak, aby \textbf{silové proudy neprocházely citlivou částí}.
	\end{itemize}
	\vfill
	{\tiny Zdroje doporučení: [11, 20, 22]; vlastní plán, bez předepsaných rozměrů nebo dělení GND.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Výkon a frekvence: oddělte rušení od citlivosti}
	\small
	\begin{itemize}
		\item Výkonový stupeň, jeho kondenzátory a připojení zátěže tvoří \textbf{kompaktní skupinu}. Velký proud nesmí sdílet úzký přívod či návrat s referencí nebo malým měřicím signálem.
		\item Měnič není vhodné odsunout daleko od vlastních kondenzátorů jen kvůli označené hranici oblasti. \textbf{Místní kritické smyčky mají přednost} před úhlednými obdélníky.
		\item Podle frekvence neseskupujte pouze součástky se stejnou hodnotou v Hz. Rozhoduje \textbf{rychlost hran, amplituda a citlivost}: i pomalu opakované PWM může mít velmi rychlé hrany.
		\item Hodiny a jejich přijímače propojte krátce nad referencí. Hodinové a silové trasy neveďte přes analogové vstupy, reference či pod jejich součástkami.
	\end{itemize}
	\vfill
	{\tiny Zdroje: [10, 16, 20, 22] TI, Analog Devices; proudové smyčky a oddělení citlivých signálů.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Teplota: blízkost podle toho, co má být teplé}
	\small
	\begin{itemize}
		\item Najděte \textbf{zdroje ztrátového výkonu}: regulátory, tranzistory, výkonové rezistory a tlumivky. Vyhraďte jim měď, prokovy, prostor pro chladič nebo proudění vzduchu.
		\item Horké součástky nehromaďte bez rozmyslu: kompaktní elektrický blok musí stále odvést teplo. Umístění u kraje či uprostřed závisí na krytu (krabičce).
		\item \textbf{Reference a přesné analogové obvody} držte mimo horká místa a prudké teplotní gradienty. Teplo se šíří i mědí a přes druhou stranu desky; mezera v osazení sama nestačí.
		\item Čidlo \textbf{okolní teploty} chraňte před vlastní elektronikou; čidlo pro \textbf{teplotu výkonového prvku} naopak tepelně přibližte sledovanému místu.
	\end{itemize}
	\vfill
	{\tiny Zdroje: [20, 23, 24] TI, Analog Devices; odvod tepla, teplotní měření a přesné obvody.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Zónování součástek neznamená rozřezání GND}
	\small
	\begin{itemize}
		\item \textbf{Funkční oblasti oddělujte rozmístěním a vedením spojů.} Výchozím řešením je souvislá GND reference. Silový návrat neveďte přes citlivý měřicí blok.
		\item Sledujte každé rozhraní mezi bloky: kudy vede signál a kudy se vrací proud? Hranice nakreslená v plánu nesmí vynutit přechod signálu přes mezeru v referenci.
		\item Galvanické oddělení nebo specifické propojení AGND/PGND řešte jen vyjímčně podle dokumentace konkrétního obvodu. Doporučuji používat vždy souvislou GND.
	\end{itemize}
	\vfill
	{\tiny Zdroje: [10, 11, 22] TI, Analog Devices; zónování a souvislá referenční plocha.}
\end{frame}

\section{\texorpdfstring{Trasy napájení}{Trasy napajeni}}
%------------------------------------------------------------------------------
\begin{frame}[t]{Napájení: krátká cesta ke kondenzátoru i zpět}
	\small
	\begin{itemize}
		\item Napájecí cesta má odpor a indukčnost. Veďte ji \textbf{krátce a dostatečně široce} s ohledem na proud, úbytek napětí a oteplení; vyhněte se úzkým hrdlům.
		\item \textbf{Blokovací kondenzátor} poskytuje místní zásobu pro rychlé změny proudu IC. Patří blízko jeho napájecího pinu a potřebuje také krátké připojení GND.
		\item \textbf{Filtrační / zásobní kondenzátor} u vstupu napájení či napájené větve nepatří na konec dlouhé odbočky. Přívod i pokračování připojte přímo k jeho ploškám.
	\end{itemize}
	\begin{block}{Pravidlo pro jednoduché trasované napájení}
	Přívod $\rightarrow$ napájecí ploška kondenzátoru $\rightarrow$ krátká větev k napájecímu pinu. Kondenzátor je připojen \textbf{mezi PWR a GND}.
	\end{block}
	\vfill
	{\tiny Zdroj doporučení: [8, 9, 11, 12] Jones, Salmony, Miller, Lakkas.}
\end{frame}

\begin{frame}[t]{Blokování: dlouhá odbočka od hlavní trasy}

		\item Proud musí procházet přes impedanci odbočky, což snižuje účinnost blokování.

%------------------------------------------------------------------------------
\begin{frame}[t]{Proč více prokovů snižuje indukčnost připojení?}
	\small
	\begin{itemize}
		\item Proud se rozdělí do \textbf{více paralelních cest}. V ideálním modelu bez vzájemné magnetické vazby platí pro $N$ stejných prokovů $L_{\text{eq}} \approx L_{\text{via}}/N$.
		\item Reálná pole prokovů se vzájemně vážou: \textbf{čtyři prokovy neznamenají automaticky čtvrtinovou indukčnost}. Důležitá je i vzdálenost GND vrstvy a poloha přívodu i návratu.
		\item Menší $L$ znamená při stejné frekvenci menší \textbf{induktivní složku impedance}: $|Z_L|=2\pi f L$. Nezaručuje nižší impedanci celé sítě při všech frekvencích.
	\end{itemize}
	\vfill
	{\tiny Zdroje: [13, 14] A. Kay (TI), Murata; paralelní model a magnetická vazba.}
\end{frame}

\section{\texorpdfstring{Spínané měniče napětí}{Spinane menice napeti}}
%------------------------------------------------------------------------------
\begin{frame}[t]{Spínací uzel: SWN}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.38\textheight,keepaspectratio]{obr/buck_switch_node.pdf}\par}
	\begin{itemize}
		\item U synchronního snižujícího měniče (buck) je \textbf{SWN společný uzel horního spínače QH, dolního QL a vstupu tlumivky L}. Spínače jsou zde znázorněné funkčními bloky.
		\item Napětí SWN rychle přechází přibližně mezi PGND a VIN; obsahuje i překmity. \textbf{Za tlumivkou je výstup VOUT}, vyhlazovaný kondenzátorem.
		\item U nesynchronního bucku nahrazuje QL dioda; do SWN patří její katoda.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [12, 16] TI; vlastní zjednodušené schéma synchronního bucku.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Proč musí být plocha SWN co nejmenší?}
	\small
	\begin{itemize}
		\item SWN patří k uzlům s \textbf{největší strmostí změny napětí} $\mathrm{d}v/\mathrm{d}t$. Rušení určují zejména rychlé hrany, nejen spínací frekvence měniče.
		\item Měď SWN má parazitní kapacitu vůči okolním spojům a plochám. Přes tuto kapacitu se při každé hraně přenáší rušivý proud:
	\end{itemize}
	\[
		i_{\text{ruš}} \approx C_{\text{par}}\,\frac{\Delta V}{t_r}
	\]
	\begin{block}{Ilustrační příklad, nikoli návrhové hodnoty}
	Změna 12 V za 3 ns dává 4 V/ns. Přes parazitní kapacitu 2 pF teče během hrany proud přibližně 8 mA.
	\end{block}
	\begin{itemize}
		\item \textbf{Větší plocha SWN obvykle zvyšuje kapacitní vazbu} a riziko EMI. Citlivá zpětná vazba či analogový vstup mohou zachytit špičky.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [16] TI; číselný příklad je vlastní výpočet pro zadané předpoklady.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{SWN na DPS: krátké a kompaktní propojení}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.35\textheight,keepaspectratio]{obr/buck_sw_layout.pdf}\par}
	\begin{itemize}
		\item \textbf{Přisuňte a natočte tlumivku k pinu SWN}; u externích spínačů seskupte i QH a QL / diodu. Měď SWN omezte na nutné propojení jejich plošek.
		\item Nepřidávejte velký polygon ani dlouhou odbočku k testovacímu bodu. SWN zbytečně nerozvádějte na další vrstvy.
		\item \textbf{Malá plocha neznamená tenký dlouhý spoj.} Zachovejte šířku pro proud a potřebný odvod tepla; rozhoduje doporučený layout konkrétního měniče.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [16, 17] TI; vlastní schematický půdorys, nikoli výrobní layout.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Malý SWN a malá proudová smyčka}
	\small
	\begin{itemize}
		\item \textbf{Plocha mědi SWN: vysoké $\mathrm{d}v/\mathrm{d}t$.} Zmenšete ji a oddalte od ní FB, kompenzaci, snímání proudu a citlivé signály; neveďte je ani přímo pod SWN.
		\item \textbf{Plocha komutační smyčky: vysoké $\mathrm{d}i/\mathrm{d}t$.} U synchronního bucku ji tvoří vstupní keramický $C_{\text{IN}}$, QH, QL a jejich spoje. Kondenzátor patří těsně k VIN a PGND výkonového stupně.
		\item Samotné zmenšení SWN neopraví dlouhou vstupní smyčku. Její parazitní indukčnost přispívá k překmitům a zvonění: $\Delta v \approx L_{\text{par}}\,\mathrm{d}i/\mathrm{d}t$.
	\end{itemize}
	\begin{block}{Postup návrhu}
	Nejprve rozmístěte výkonový stupeň a vstupní C, poté tlumivku a kompaktní SWN. Zpětnou vazbu odebírejte z výstupu za tlumivkou. Pro GND, případný keepout pod SWN a chlazení respektujte datasheet a vzorovou desku.
	\end{block}
	\vfill
	{\tiny Zdroj doporučení: [12, 16, 17] TI; popsaná komutační smyčka platí pro buck.}
\end{frame}

%------------------------------------------------------------------------------
\section{\texorpdfstring{Zatížitelnost tras}{Zatizitelnost tras}}
%------------------------------------------------------------------------------
\begin{frame}[t]{Zatížitelnost: proud, průřez a oteplení}
	\small
	\begin{itemize}
		\item Neexistuje jedna univerzální hodnota \textbf{A na mm šířky}. Rozhoduje šířka $w$, tloušťka mědi $t$, vrstva, okolní měď, chlazení i teplota okolí.
		\item Pro trvalé zatížení stanovte \textbf{dovolené oteplení nad okolí} $\Delta T$. Například při okolí 50\,$^\circ$C a oteplení 20\,$^\circ$C má spoj přibližně 70\,$^\circ$C.
		\item Ztráty určujte z \textbf{efektivního proudu} $I_{\text{RMS}}$, nikoli jen z průměru. U impulzů sledujte i špičku, dobu trvání a opakování.
	\end{itemize}
	\[
		R=\rho\frac{l}{w t},\qquad \Delta U=I R,\qquad P=I_{\text{RMS}}^2 R
	\]
	\begin{block}{Kontrolujte celou proudovou cestu}
	Přívod i návrat, úzká místa polygonů, připojení plošek, prokovy a paprsky termálních odlehčení. Výrobní minimum šířky není zárukou proudové zatížitelnosti.
	\end{block}
	\vfill
	{\tiny Zdroje: [17, 18, 20] TI, IPC-2152; vztahy pro odpor a Jouleovy ztráty.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Návrh šířky: kalkulátor a kontrola úbytku}
	\footnotesize
	\begin{enumerate}
		\item Zadejte proud, dovolené $\Delta T$, skutečnou tloušťku mědi a vrstvu. Pro školní příklad lze uvažovat 35\,$\mu$m, ale \textbf{ověřte skladbu u výrobce}.
		\item Pro návrh šířky použijte \textbf{kalkulátor v KiCadu s jednodušším modelem podle IPC-2221}. V Kalkulačních nástrojích vyberte šířku spoje; pro prokovy je zde samostatný nástroj.
		\item \textbf{IPC-2152 popisuje zatížitelnost podrobněji}; pro vaše práce použijte dostupný kalkulátor v KiCadu. Výsledek berte jako odhad a ověřte také úbytek napětí a ztráty.
	\end{enumerate}
	\medskip
	{\centering

%------------------------------------------------------------------------------
\begin{frame}[t]{Prokov: proud vede pokovená stěna otvoru}
	\small
	\begin{itemize}
		\item Běžný prokov není plný měděný drát. Rozhoduje \textbf{průměr hotového otvoru $d$, tloušťka pokovení $t_p$ a délka proudové cesty $h$}.
		\item Vnější průměr plošky a mezikruží jsou důležité pro připojení a výrobu, ale samotné zvětšení plošky nezvětší průřez stěny.
	\end{itemize}
	\[
		A_{\text{stěny}}=\pi(d+t_p)t_p,\qquad R_{\text{prokovu}}\approx\rho\frac{h}{A_{\text{stěny}}}
	\]
	\begin{block}{Ilustrační odporový model při 20\,$^\circ$C}
	Pro $d=0{,}30$ mm, $t_p=25\,\mu$m a $h=1{,}6$ mm vychází $A\approx0{,}0255$ mm$^2$ a $R\approx1{,}08$ m$\Omega$. Pokovení 25\,$\mu$m je zde předpoklad, nikoli automatická vlastnost prokovu.
	\end{block}
	\textbf{Z odporu samotného nelze určit dovolený proud.} Záleží i na odvodu tepla do vrstev a přípustné teplotě.
	\vfill
	{\tiny Zdroje: [19, 20] KiCad, TI; vlastní model homogenní měděné stěny bez odporu přívodů.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Větší proud: více prokovů a široké připojení}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.30\textheight,keepaspectratio]{obr/current_via_array.pdf}\par}
	\begin{itemize}
		\item Pro přechod silové cesty mezi vrstvami použijte \textbf{skupinu prokovů s krátkým širokým přívodem i odběrem}. Jediné společné úzké hrdlo jejich přínos omezí.
		\item Pro $N$ stejných rovnoměrně zatížených prokovů je ideálně $R_{\text{eq}}\approx R_{\text{prokovu}}/N$. Reálné proudy závisí na rozmístění a odporu okolní mědi.
		\item \textbf{Čtyři prokovy automaticky nezaručují čtyřnásobný dovolený proud}: sdílejí tepelné okolí a nemusí nést stejný proud. Počet volte s rezervou a ověřte oteplení.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [20] TI, část 3; vlastní schematické půdorysy, nikoli výrobní rozměry.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Necking: krátké místní zúžení trasy}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.32\textheight,keepaspectratio]{obr/current_necking.pdf}\par}
	\begin{itemize}
		\item \textbf{Necking / neck-down} je místní zúžení široké trasy při vstupu na malou plošku nebo průchodu mezi ploškami. Po průchodu se vraťte k návrhové šířce.
		\item Úzký úsek ponechte \textbf{jen tak dlouhý, jak je nutné}, a přejděte plynule do širší mědi. Zachovejte výrobní šířku a izolační odstupy.
		\item Zúžením teče stejný proud jako zbytkem větve. Krátký necking \textbf{může} být přijatelný díky malé délce a odvodu tepla do okolní mědi.
	\end{itemize}
	\vfill
	{\tiny Zdroje: [18, 21] IPC, příklad místní změny šířky TI; vlastní obecná ilustrace.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Necking: posuďte šířku i délku}
	\small
	Pro zúžení šířky 0,25 mm v mědi 35\,$\mu$m a DC proud 2 A:
	\medskip
	{\centering
	\begin{tabular}{c|r|r|r}
		Délka zúžení & Odpor & Úbytek & Ztráta\\\hline
		0,5 mm & 0,983 m$\Omega$ & 1,97 mV & 3,93 mW\\
		5,0 mm & 9,83 m$\Omega$ & 19,7 mV & 39,3 mW
	\end{tabular}\par}
	\medskip
	\begin{itemize}
		\item Při stejném průřezu znamená desetkrát delší úsek \textbf{desetkrát větší odpor a celkový ztrátový výkon}. Teplotu neurčuje tento poměr sám; rozhoduje i odvod tepla.
		\item Pro kritickou cestu ověřte oteplení výpočtem (KiCad) / simulací nebo měřením při nejhorším provozním zatížení.
		\item Pokud výsledek nevyhoví, změňte rozmístění, zkraťte zúžení nebo upravte připojení. \textbf{DRC samo neověřuje tepelnou zatížitelnost.}
	\end{itemize}
	\vfill
	{\tiny Zdroje: [18, 19, 21]; vlastní výpočet s $\rho=0{,}0172\,\Omega\,\text{mm}^2/\text{m}$ při 20\,$^\circ$C. Nejde o proudový limit.}
\end{frame}
