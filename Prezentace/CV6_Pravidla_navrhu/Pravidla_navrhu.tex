\begin{frame}[t]{Stackup: nejde pouze o počet vrstev}
	\begin{itemize}
		\item \textbf{Stackup} určuje pořadí měděných vrstev, jejich funkci a tloušťky izolačních materiálů mezi nimi.
		\item \textbf{SIG}: signály; \textbf{PWR}: napájení; \textbf{GND}: referenční vodič a návratová cesta proudu.
		\item Zápis \textbf{SIG + PWR} označuje jednu společnou vrstvu pro signálové trasy a rozvod napájení, nikoli dvě vrstvy.
		\item Pořadí v obrázcích čteme \textbf{shora dolů}. Čísla L1--L4 odpovídají pořadí měděných vrstev.
	\end{itemize}
	\begin{block}{Základní doporučení}
	Signál potřebuje blízkou a souvislou referenční plochu. Důležité jsou i rychlost hran signálu a délka vedení, nejen hodinová frekvence.
	\end{block}
	\vfill
	{\tiny Zdroj doporučení: [1, 5, 6]. Úplný seznam zdrojů je na konci prezentace.}
\end{frame}

\begin{frame}[t]{Dvě vrstvy: výhody a omezení}
	\begin{itemize}
		\item \textbf{Výhody:} jednoduchá a levná výroba; přehledná návratová cesta oproti desce s rozdělenou zemí na obou stranách.
		\item \textbf{Nevýhody:} méně prostoru pro signály a napájení; u běžné tloušťky 1,6 mm je protější GND vzdálená, takže vazba a řízení impedance jsou horší než u vhodné čtyřvrstvé desky.
		\item \textbf{Použití:} jednodušší obvody s krátkými vedeními. Spodní GND co nejméně přerušovat; trasy přes celou spodní stranu mohou rozříznout návratovou cestu.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [1] Z. Peterson.}
\end{frame}

\begin{frame}[t]{SIG / GND / PWR / SIG: výhody a omezení}
	\begin{itemize}
		\item \textbf{Výhody:} dvě signálové vrstvy a samostatný rozvod napájení; horní signály mají blízkou GND.
		\item \textbf{Nevýhody:} spodní signály mají nejbližší referenci PWR. Rozdělení napájecí plochy a změna reference při přechodu vrstev komplikují návrat proudu.
		\item \textbf{Použití:} rychlé signály přednostně na L1 nad GND. Plná PWR může být referencí pro L4, ale je nutné vyřešit návaznost návratového proudu na GND.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [2] Z. Peterson.}

\begin{frame}[t]{Dvě plochy GND: výhody a omezení}
	\begin{itemize}
		\item \textbf{Výhody:} obě vnější vrstvy mají blízkou referenci GND; vnitřní plochy oddělují signály na protilehlých stranách.
		\item \textbf{Nevýhody:} napájení sdílí prostor se signály, není k dispozici samostatná PWR vrstva. Pro větší proudy a mnoho napěťových úrovní (zdrojů) může chybět místo.
		\item \textbf{Doporučení:} vhodný výchozí stackup pro běžnou čtyřvrstvou digitální desku. GND plochy propojit; u přechodu signálu vložit blízký GND prokov.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [3] Z. Peterson.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Návratová cesta při změně signálové vrstvy}
	\begin{itemize}
		\item U rychlých změn proudu se návratový proud soustřeďuje do blízké referenční plochy pod trasou.
		\item \textbf{GND → GND:} při přechodu L1 $\leftrightarrow$ L4 propojte obě referenční plochy GND prokovem poblíž signálového prokovu.
		\item \textbf{GND → PWR:} samotný GND prokov nestačí. Přechod návratového proudu vyžaduje vhodnou vazbu mezi PWR a GND, např. blízký blokovací kondenzátor s krátkým připojením.
		\item Signál neveďte přes mezeru, výřez nebo rozhraní různých napětí v jeho referenční ploše.
	\end{itemize}
	\begin{block}{Výhoda dvou GND vrstev}
	Návratovou cestu lze při přechodu jednodušeji navázat GND prokovem. GND a PWR se naopak nesmějí spojit přímým prokovem.
	\end{block}
	\vfill
	{\tiny Zdroj doporučení: [4, 5, 7, 9] Peterson, Bogatin, Kay, Salmony.}
\end{frame}

\begin{frame}[t]{Tloušťky dielektrika a napájení}
	\begin{itemize}
		\item U čtyřvrstvé desky požadujte blízké dvojice \textbf{L1--L2} a \textbf{L3--L4}; běžná stavba má tenké prepregy a silnější středové jádro.
		\item Menší vzdálenost signálu od reference pomáhá omezit plochu proudové smyčky. Impedanci ale určuje i šířka a tloušťka mědi a vlastnosti dielektrika.
		\item Vzdálené plochy PWR a GND mají malou vzájemnou kapacitu. \textbf{Samostatná PWR vrstva nenahrazuje blokovací kondenzátory.}
		\item Konkrétní tloušťky, materiály a dostupné varianty ověřte u výrobce \textbf{před návrhem impedančně řízených spojů}.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [2, 6] Z. Peterson, Phil's Lab.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Jak zvolit výchozí skladbu?}
	\small
	\begin{description}
		\item[(SIG + PWR) / GND] Pro jednoduché, méně husté desky. Prioritou je zachování GND, ne zaplnění obou stran signály.
		\item[SIG / GND / PWR / SIG] Když pomůže samostatná napájecí plocha a nejdůležitější signály zůstanou nad GND. Není to automaticky chybné řešení, ale spodní signály vyžadují pozornost.
		\item[(SIG + PWR) / GND / GND / (SIG + PWR)] Vhodný výchozí kompromis pro digitální signály na obou stranách. Napájení řešte dostatečně širokými trasami či polygony.
	\end{description}
	\begin{block}{Neexistuje jedna skladba pro všechny desky}
	Rozhodují proudy, napětí, rychlost hran, hustota zapojení a požadavky na rušení. Pokud se napájení a signály nevejdou bez porušení návratových cest, zvažte více vrstev.
	\end{block}
	\vfill
	{\tiny Zdroj doporučení: [1--6]. Výukové shrnutí; úplný seznam zdrojů následuje na konci.}
\end{frame}

\begin{frame}[t]{Funkční slot: deska není proříznutá, GND ano}
	\small
	\begin{itemize}
		\item Zde výrazem \textbf{funkční slot} označujeme mezeru v referenční mědi vytvořenou jinými prvky návrhu, nikoli mechanické frézování DPS.
		\item Vzniká například spojením \textbf{izolačních odstupů kolem cizích plošek} konektoru nebo vedením \textbf{jiné sítě uvnitř GND plochy}.
		\item Signál může být správně propojený, ale jeho vysokofrekvenční návrat musí mezeru obejít. Zvětšuje se proudová smyčka a riziko rušení i změny impedance.
	\end{itemize}
	\begin{block}{Sledujeme dva vodiče, ne jen signál}
	Průchod mezi piny ani křížení na různých vrstvách nejsou samy o sobě chybou. Rozhodující je, zda pod signálem zůstává dostatečně souvislá referenční měď.
	\end{block}
	\vfill
	{\tiny Zdroj doporučení: [1, 9, 10] Peterson, Salmony, Texas Instruments.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Konektor: signál projde, návrat musí oklikou}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.50\textheight,keepaspectratio]{obr/slot_connector_bad.pdf}\par}
	\begin{itemize}
		\item Trasa na F.Cu se vejde mezi piny. Odstupy cizích průchozích plošek však na B.Cu splývají do \textbf{souvislé mezery v GND}.
		\item Návrat nemůže sledovat signál přímo; musí obejít řadu plošek. Případně ponechané úzké hrdlo mědi může být také nevhodné z pohledu impedance.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [9, 10] P. Salmony, Texas Instruments.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Konektor: veďte signál nad souvislou GND}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.52\textheight,keepaspectratio]{obr/slot_connector_better.pdf}\par}
	\begin{itemize}
		\item Zde signál obchází konec řady plošek \textbf{nad souvislou GND}; návrat může zůstat blízko jeho trasy.
		\item Jinou možností je změnit rozmístění či pinout a přidělit vhodné piny GND. Výsledek posuzujte v \textbf{naplněné referenční ploše}, ne jen podle spojnic.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [9, 10] P. Salmony, Texas Instruments.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Dvě vrstvy: spodní trasa rozřízne GND}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.52\textheight,keepaspectratio]{obr/slot_crossing_bad.pdf}\par}
	\begin{itemize}
		\item Horní signál kříží v půdorysu jinou trasu na B.Cu. \textbf{Spodní trasa i její odstupy vytlačí GND} právě pod horním signálem.
		\item Dlouhá spodní trasa vynutí dlouhou okliku návratu. Pokud oddělí GND až od okraje k okraji, nemusí zbýt ani vodivá cesta v této vrstvě.
	\end{itemize}

%------------------------------------------------------------------------------
\begin{frame}[t]{Křížení: blízký GND můstek jako kompromis}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.50\textheight,keepaspectratio]{obr/slot_crossing_better.pdf}\par}
	\begin{itemize}
		\item Nejprve zkuste spodní trasu \textbf{přemístit nebo zkrátit} tak, aby nepřerušovala referenci důležitého signálu.
		\item Kompromis: blízké GND prokovy a krátký GND můstek na F.Cu přenesou návrat přes mezeru. \textbf{Nespojí cizí síť s GND}; stále přidávají indukčnost a nenahrazují souvislou plochu.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [1, 9, 10] Peterson, Salmony, Texas Instruments.}
\end{frame}

%------------------------------------------------------------------------------
\section{\texorpdfstring{Trasy napájení}{Trasy napajeni}}
%------------------------------------------------------------------------------
\begin{frame}[t]{Napájení: krátká cesta ke kondenzátoru i zpět}
	\small
	\begin{itemize}
		\item Napájecí cesta má odpor a indukčnost. Veďte ji \textbf{krátce a dostatečně široce} s ohledem na proud, úbytek napětí a oteplení; vyhněte se úzkým hrdlům.
		\item \textbf{Blokovací kondenzátor} poskytuje místní zásobu pro rychlé změny proudu IC. Patří blízko jeho napájecího pinu a potřebuje také krátké připojení GND.
		\item \textbf{Filtrační / zásobní kondenzátor} u vstupu napájení či napájené větve nepatří na konec dlouhé odbočky. Přívod i pokračování připojte přímo k jeho ploškám.
	\end{itemize}
	\begin{block}{Pravidlo pro jednoduché trasované napájení}
	Přívod $\rightarrow$ napájecí ploška kondenzátoru $\rightarrow$ krátká větev k napájecímu pinu. Kondenzátor je stále \textbf{mezi PWR a GND}, ne v sérii s napájením.
	\end{block}
	\vfill
	{\tiny Zdroj doporučení: [8, 9, 11, 12] Jones, Salmony, Miller, Lakkas.}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Blokování: dlouhá odbočka od hlavní trasy}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.50\textheight,keepaspectratio]{obr/power_decoupling_bad.pdf}\par}
	\begin{itemize}
		\item U1 i U2 jsou připojené přímo k hlavní trase. C1 a C2 jsou elektricky připojené, ale \textbf{přes dlouhé T odbočky}.
		\item Rychlý proud z kondenzátoru musí projít dlouhou přípojkou k IC. Samotná správná kapacita tak nezaručí účinné blokování.
	\end{itemize}
	\vfill
	{\tiny Zdroj doporučení: [8, 11] D. L. Jones, D. Miller (TI).}
\end{frame}

%------------------------------------------------------------------------------
\begin{frame}[t]{Blokování: odbočení na plošce kondenzátoru}
	\small
	{\centering
	\includegraphics[width=\linewidth,height=.50\textheight,keepaspectratio]{obr/power_decoupling_better.pdf}\par}
	\begin{itemize}
		\item Hlavní napájecí trasa prochází \textbf{napájecími ploškami C1 a C2}. Z každé plošky vede krátká vlastní větev k příslušnému IC.
		\item Kondenzátor umístěte co nejblíže napájecímu pinu; GND připojte krátce do souvislé plochy. \textbf{Tvar T nevadí, pokud uzel leží přímo na plošce.}

%------------------------------------------------------------------------------
\begin{frame}[t]{Proč více prokovů snižuje indukčnost připojení?}
	\small
	\begin{itemize}
		\item Proud se rozdělí do \textbf{více paralelních cest}. V ideálním modelu bez vzájemné magnetické vazby platí pro $N$ stejných prokovů $L_{\text{eq}} \approx L_{\text{via}}/N$.
		\item Reálná pole prokovů se vzájemně vážou: \textbf{čtyři prokovy neznamenají automaticky čtvrtinovou indukčnost}. Důležitá je i vzdálenost GND vrstvy a poloha přívodu i návratu.
		\item Menší $L$ znamená při stejné frekvenci menší \textbf{induktivní složku impedance}: $|Z_L|=2\pi f L$. Nezaručuje nižší impedanci celé sítě při všech frekvencích.
	\end{itemize}
	\begin{block}{Magnetické vysvětlení přesněji}
	Indukčnost určuje vázaný magnetický tok na jednotku proudu, ne samotná délka jedné indukční čáry. Posuzujeme rozložení pole a geometrii \textbf{celé proudové smyčky}.
	\end{block}
	\vfill
	{\tiny Zdroje: [13, 14] A. Kay (TI), Murata; paralelní model a magnetická vazba.}
\end{frame}
